FFmpeg configure:
  --prefix=<prefix>
  --arch=arm64
  --target-os=darwin
  --enable-cross-compile
  --cc=clang
  --cxx=clang++
  --pkg-config-flags=--static
  --enable-static
  --disable-shared
  --disable-autodetect
  --disable-network
  --disable-doc
  --disable-ffplay
  --disable-debug
  --enable-videotoolbox
  --enable-audiotoolbox
  --enable-zlib
  --enable-libsvtav1
  --enable-libdav1d
  --enable-libvmaf
  --enable-libzimg
  --enable-libopus
  --enable-libmp3lame
  --enable-libvorbis
  --extra-cflags=-arch arm64 -mmacosx-version-min=13.0 -O2 -I<prefix>/include
  --extra-ldflags=-arch arm64 -mmacosx-version-min=13.0 -L<prefix>/lib
  --extra-libs=-lc++
